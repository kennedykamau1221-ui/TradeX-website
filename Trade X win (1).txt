export function ema(a,p){if(a.length<p)return[];let k=2/(p+1),e=a.slice(0,p).reduce((x,y)=>x+y,0)/p,o=Array(p-1).fill(null);o.push(e);for(let i=p;i<a.length;i++){e=a[i]*k+e*(1-k);o.push(e)}return o}
export function rsi(a,p=14){if(a.length<=p)return[];let g=0,l=0;for(let i=1;i<=p;i++){let d=a[i]-a[i-1];g+=Math.max(d,0);l+=Math.max(-d,0)}let ag=g/p,al=l/p,o=Array(p).fill(null);o.push(al?100-100/(1+ag/al):100);for(let i=p+1;i<a.length;i++){let d=a[i]-a[i-1];ag=(ag*(p-1)+Math.max(d,0))/p;al=(al*(p-1)+Math.max(-d,0))/p;o.push(al?100-100/(1+ag/al):100)}return o}
export function atr(c,p=14){if(c.length<=p)return[];let t=c.map((x,i)=>i?Math.max(n(x.high)-n(x.low),Math.abs(n(x.high)-n(c[i-1].close)),Math.abs(n(x.low)-n(c[i-1].close))):n(x.high)-n(x.low));let a=t.slice(0,p).reduce((x,y)=>x+y,0)/p,o=Array(p-1).fill(null);o.push(a);for(let i=p;i<t.length;i++){a=(a*(p-1)+t[i])/p;o.push(a)}return o}
function piv(c){let hi=[],lo=[];for(let i=2;i<c.length-2;i++){let h=n(c[i].high),l=n(c[i].low),H=true,L=true;for(let j=i-2;j<=i+2;j++)if(j!==i){if(n(c[j].high)>=h)H=false;if(n(c[j].low)<=l)L=false}if(H)hi.push({i,p:h});if(L)lo.push({i,p:l})}return{hi,lo}}
export function analyze(raw){let c=raw.map(x=>({...x,open:n(x.open),high:n(x.high),low:n(x.low),close:n(x.close)})),cl=c.map(x=>x.close),e20=ema(cl,20),e50=ema(cl,50),e200=ema(cl,200),R=rsi(cl),A=atr(c),p=piv(c),last=c.at(-1),h=p.hi.at(-1),ph=p.hi.at(-2),l=p.lo.at(-1),pl=p.lo.at(-2);let bias="RANGE";if(h&&ph&&l&&pl)bias=h.p>ph.p&&l.p>pl.p?"BULLISH":h.p<ph.p&&l.p<pl.p?"BEARISH":"RANGE";if(e20.at(-1)&&e50.at(-1)&&e200.at(-1))if(e20.at(-1)>e50.at(-1)&&e50.at(-1)>e200.at(-1))bias="BULLISH";else if(e20.at(-1)<e50.at(-1)&&e50.at(-1)<e200.at(-1))bias="BEARISH";let event=h&&last.close>h.p?"BOS_UP":l&&last.close<l.p?"BOS_DOWN":"NONE";let f=[],ob=[];for(let i=2;i<c.length;i++){if(c[i].low>c[i-2].high)f.push({type:"BULLISH_FVG",low:c[i-2].high,high:c[i].low});if(c[i].high<c[i-2].low)f.push({type:"BEARISH_FVG",low:c[i].high,high:c[i-2].low});let a=c[i-1],b=c[i];if(a.close<a.open&&b.close>b.open&&b.close>a.high)ob.push({type:"BULLISH_OB",low:a.low,high:a.high});if(a.close>a.open&&b.close<b.open&&b.close<a.low)ob.push({type:"BEARISH_OB",low:a.low,high:a.high})}let score=(bias==="BULLISH"?1:bias==="BEARISH"?-1:0)+(R.at(-1)>55?1:R.at(-1)<45?-1:0)+(event==="BOS_UP"?1:event==="BOS_DOWN"?-1:0);return{price:last.close,bias,event,recommendation:score>=2?"LONG BIAS":score<=-2?"SHORT BIAS":"WAIT / RANGE",score,indicators:{ema20:e20.at(-1),ema50:e50.at(-1),ema200:e200.at(-1),rsi14:R.at(-1),atr14:A.at(-1)},fvg:f.slice(-8),orderBlocks:ob.slice(-8),support:p.lo.slice(-5).map(x=>x.p),resistance:p.hi.slice(-5).map(x=>x.p),note:"Rule-based analysis; not a guarantee."}}

}TradeX Website — combined starter build
Responsive website form for desktop and Android browsers. Includes a market dashboard, provider-backed quote/analysis endpoints, candlestick chart, technical-analysis summaries, a local demo-trading simulator, demo wallet request forms, and a protected administrator risk-settings panel.
What works in this build
Website UI: Overview, Markets, Trade, Wallet, Analysis and Admin sections.
Provider-backed read-only quotes and candles for XAU/USD, EUR/USD, GBP/USD and BTC/USD when a valid Twelve Data API key is configured.
RSI/EMA/structure/FVG/order-block summaries from the included analysis module.
Demo trades use virtual funds and browser-local storage only.
Deposit/withdrawal forms create local demo requests only; they do not process payments.
MT5 bridge source is included for future integration/testing, but live trading is disabled by default.
Run locally
Install Node.js 20+.
Copy .env.example to .env.
Set a long random ADMIN_TOKEN and BRIDGE_TOKEN; add a valid TWELVE_DATA_API_KEY if you have one.
For local use set PUBLIC_URL=http://localhost:8787 and ALLOWED_ORIGINS=http://localhost:8787 (or leave ALLOWED_ORIGINS blank for development).
Run npm install then npm start.
Open http://localhost:8787.
Without a market-data API key, the website and demo simulator still open, but live quotes and candle analysis will show a setup message.
Deploy as a website
Deploy this Node.js project to a host that supports Node.js.
Configure the environment variables from .env.example in the hosting dashboard.
Set PUBLIC_URL to your HTTPS website address and ALLOWED_ORIGINS to that same origin.
Use HTTPS, a strong admin/bridge token, protected server logs, backups, and a managed database before public production use.
Keep .env, API keys, broker credentials and MT5 credentials out of source control and browser code.
Trading and payment limits
This is a website starter/prototype, not a licensed broker or a payment processor. Demo trades and wallet requests are browser-local and are not real financial transactions. Real deposits, withdrawals, customer accounts, synthetic indices and live orders require secure server-side account/ledger implementation, a suitable authorized provider, testing, and legal/regulatory review. Live MT5 execution remains disabled by default. Never use a simulated market as if it were a real provider market.
MT5 bridge
TradeXBridge.mq5 is included for controlled testing with MT5. Do not enable real trading until the bridge, broker permissions, risk controls, deployment security, and legal requirements have been independently reviewed. Website deployment quick guide
Upload this project to a Node.js host or deploy the ZIP using the host's supported workflow.
Configure environment variables from .env.example in the host dashboard. Do not upload a populated .env file.
Set PUBLIC_URL=https://your-real-domain.example and ALLOWED_ORIGINS=https://your-real-domain.example.

Confirm the site is served over HTTPS and test in mobile Chrome.
Demo trading is local-only; deposits and withdrawals are request-form demos. Real-money services are not enabled by this package.
The package cannot create a public domain or hosting account for you. You will need to deploy it to a hosting provider and attach a domain to obtain a permanent public website URL.
